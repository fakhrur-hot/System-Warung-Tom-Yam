Here is a modular architecture using Kotlin and Jetpack Compose for your POS app.

It includes:

Data Models & State for payment methods and API configuration.

Admin Settings UI (enable/disable toggles + API credential fields).

Gateway Payment Request Service (supporting DuitNow QR, TNG, GrabPay, Boost, ShopeePay, FPX, MAE, Cards).

Checkout & Payment Screen (Grand Total, enabled payment options grid, and checkout trigger).

1. Domain Models & Configuration State
First, define your payment methods as an enum and create a state data class for API credentials and toggles.

Kotlin
enum class PaymentMethod(
    val id: String,
    val displayName: String,
    val category: Category
) {
    DUITNOW_QR("DUITNOW_QR", "DuitNow QR", Category.QR_PAYNET),
    TNG("TNG", "Touch 'n Go eWallet", Category.E_WALLET),
    GRABPAY("GRABPAY", "GrabPay", Category.E_WALLET),
    BOOST("BOOST", "Boost", Category.E_WALLET),
    SHOPEEPAY("SHOPEEPAY", "ShopeePay", Category.E_WALLET),
    MAYBANK_MAE("MAE", "Maybank MAE QRPay", Category.E_WALLET),
    FPX("FPX", "FPX Online Banking", Category.ONLINE_BANKING),
    CREDIT_DEBIT("CARD", "Credit / Debit Card", Category.CARD);

    enum class Category { QR_PAYNET, E_WALLET, ONLINE_BANKING, CARD }
}

// Model for saving Admin API credentials and method toggles
data class GatewayConfig(
    val merchantId: String = "",
    val apiKey: String = "",
    val secretKey: String = "",
    val isSandbox: Boolean = true,
    // Toggles for enabling/disabling methods on POS
    val enabledMethods: Map<PaymentMethod, Boolean> = PaymentMethod.values().associateWith { true }
)
2. Admin Settings Screen (Toggles & Credentials)
This Compose UI allows the manager/admin to enter API credentials and toggle available payment channels on the POS device.

Kotlin
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.text.input.PasswordVisualTransformation
import androidx.compose.ui.unit.dp

@Composable
fun AdminPaymentSettingsScreen(
    currentConfig: GatewayConfig,
    onSaveConfig: (GatewayConfig) -> Unit
) {
    var merchantId by remember { mutableStateOf(currentConfig.merchantId) }
    var apiKey by remember { mutableStateOf(currentConfig.apiKey) }
    var secretKey by remember { mutableStateOf(currentConfig.secretKey) }
    var isSandbox by remember { mutableStateOf(currentConfig.isSandbox) }
    val enabledMethods = remember { mutableStateMapOf<PaymentMethod, Boolean>().apply { putAll(currentConfig.enabledMethods) } }

    Column(modifier = Modifier.fillMaxSize().padding(16.dp)) {
        Text("Payment Gateway Configuration", style = MaterialTheme.typography.headlineMedium)
        Spacer(modifier = Modifier.height(16.dp))

        // --- SECTION 1: API CREDENTIALS ---
        Card(modifier = Modifier.fillMaxWidth().padding(bottom = 16.dp)) {
            Column(modifier = Modifier.padding(16.dp)) {
                Text("API Credentials", style = MaterialTheme.typography.titleMedium)
                
                OutlinedTextField(
                    value = merchantId,
                    onValueChange = { merchantId = it },
                    label = { Text("Merchant ID") },
                    modifier = Modifier.fillMaxWidth()
                )
                OutlinedTextField(
                    value = apiKey,
                    onValueChange = { apiKey = it },
                    label = { Text("API Key") },
                    modifier = Modifier.fillMaxWidth()
                )
                OutlinedTextField(
                    value = secretKey,
                    onValueChange = { secretKey = it },
                    label = { Text("Secret Key") },
                    visualTransformation = PasswordVisualTransformation(),
                    modifier = Modifier.fillMaxWidth()
                )
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Checkbox(checked = isSandbox, onCheckedChange = { isSandbox = it })
                    Text("Sandbox / Test Mode")
                }
            }
        }

        // --- SECTION 2: ENABLE / DISABLE PAYMENT METHODS ---
        Text("Enabled Payment Channels", style = MaterialTheme.typography.titleMedium)
        
        LazyColumn(modifier = Modifier.weight(1f)) {
            items(PaymentMethod.values()) { method ->
                Row(
                    modifier = Modifier.fillMaxWidth().padding(vertical = 4.dp),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Column {
                        Text(method.displayName, style = MaterialTheme.typography.bodyLarge)
                        Text(method.category.name, style = MaterialTheme.typography.bodySmall)
                    }
                    Switch(
                        checked = enabledMethods[method] ?: false,
                        onCheckedChange = { isChecked -> enabledMethods[method] = isChecked }
                    )
                }
                Divider()
            }
        }

        Button(
            onClick = {
                onSaveConfig(
                    GatewayConfig(
                        merchantId = merchantId,
                        apiKey = apiKey,
                        secretKey = secretKey,
                        isSandbox = isSandbox,
                        enabledMethods = enabledMethods
                    )
                )
            },
            modifier = Modifier.fillMaxWidth().height(50.dp)
        ) {
            Text("Save Settings")
        }
    }
}
3. Gateway Payment Processor
This service accepts the total amount (in MYR cents), order details, and selected payment channel, routing the payload to your payment gateway aggregator (Revenue Monster, iPay88, RMS, or Curlec).

Kotlin
import kotlinx.serialization.Serializable

@Serializable
data class PosCheckoutPayload(
    val merchantId: String,
    val orderId: String,
    val amountCents: Long, // e.g., 2550 for RM 25.50
    val paymentMethodCode: String,
    val currency: String = "MYR",
    val customerAuthCode: String? = null // Passed if POS camera scans TNG/MAE barcode
)

@Serializable
data class GatewayPaymentResult(
    val success: Boolean,
    val transactionId: String?,
    val qrString: String?, // Payload to generate DuitNow QR on POS screen
    val checkoutUrl: String?, // Hosted web view URL for Card / FPX redirection
    val errorMessage: String? = null
)

class PaymentGatewayClient(private val config: GatewayConfig) {

    suspend fun initiatePayment(
        orderId: String,
        totalAmountCents: Long,
        selectedMethod: PaymentMethod,
        customerBarcode: String? = null
    ): GatewayPaymentResult {
        val payload = PosCheckoutPayload(
            merchantId = config.merchantId,
            orderId = orderId,
            amountCents = totalAmountCents,
            paymentMethodCode = selectedMethod.id,
            customerAuthCode = customerBarcode
        )

        // Switch execution based on payment method category
        return when (selectedMethod) {
            PaymentMethod.DUITNOW_QR,
            PaymentMethod.TNG,
            PaymentMethod.GRABPAY,
            PaymentMethod.BOOST,
            PaymentMethod.SHOPEEPAY,
            PaymentMethod.MAYBANK_MAE -> {
                if (customerBarcode != null) {
                    // Merchant Scan Flow (POS Scans Customer's Barcode)
                    executeDirectScanPayment(payload)
                } else {
                    // Customer Scan Flow (POS Displays Dynamic QR Code)
                    generateDynamicQr(payload)
                }
            }
            PaymentMethod.FPX,
            PaymentMethod.CREDIT_DEBIT -> {
                // Online Banking / Card Reader Terminal Trigger
                executeCardOrFpxPayment(payload)
            }
        }
    }

    private suspend fun generateDynamicQr(payload: PosCheckoutPayload): GatewayPaymentResult {
        // Mock gateway call returning dynamic DuitNow/eWallet QR string
        return GatewayPaymentResult(
            success = true,
            transactionId = "TXN_${System.currentTimeMillis()}",
            qrString = "00020101021226580014my.com.paynet.duitnow...", // DuitNow Payload
            checkoutUrl = null
        )
    }

    private suspend fun executeDirectScanPayment(payload: PosCheckoutPayload): GatewayPaymentResult {
        // Mock instant barcode authorization
        return GatewayPaymentResult(
            success = true,
            transactionId = "TXN_${System.currentTimeMillis()}",
            qrString = null,
            checkoutUrl = null
        )
    }

    private suspend fun executeCardOrFpxPayment(payload: PosCheckoutPayload): GatewayPaymentResult {
        return GatewayPaymentResult(
            success = true,
            transactionId = "TXN_${System.currentTimeMillis()}",
            qrString = null,
            checkoutUrl = "https://gateway.com.my/checkout/${payload.orderId}"
        )
    }
}
4. POS Checkout Screen (Grand Total & Payment Trigger)
This UI dynamically renders only enabled payment options saved in the admin configuration and displays the formatted grand total.

Kotlin
import androidx.compose.foundation.border
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.grid.GridCells
import androidx.compose.foundation.lazy.grid.LazyVerticalGrid
import androidx.compose.foundation.lazy.grid.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp
import java.util.Locale

@Composable
fun PosCheckoutScreen(
    orderId: String,
    subtotalCents: Long,
    taxCents: Long,
    config: GatewayConfig,
    onPaymentSuccess: (String) -> Unit
) {
    val grandTotalCents = subtotalCents + taxCents
    val formattedTotal = String.format(Locale.US, "RM %.2f", grandTotalCents / 100.0)
    
    // Filter channels enabled in Admin settings
    val availableMethods = remember(config) {
        PaymentMethod.values().filter { config.enabledMethods[it] == true }
    }
    
    var selectedMethod by remember { mutableStateOf<PaymentMethod?>(availableMethods.firstOrNull()) }
    var isProcessing by remember { mutableStateOf(false) }

    Column(modifier = Modifier.fillMaxSize().padding(24.dp)) {
        
        // --- GRAND TOTAL SUMMARY ---
        Card(
            colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.primaryContainer),
            modifier = Modifier.fillMaxWidth().padding(bottom = 24.dp)
        ) {
            Column(modifier = Modifier.padding(20.dp)) {
                Text("Order #$orderId", style = MaterialTheme.typography.titleMedium)
                Spacer(modifier = Modifier.height(8.dp))
                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
                    Text("Subtotal:")
                    Text(String.format(Locale.US, "RM %.2f", subtotalCents / 100.0))
                }
                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
                    Text("SST / Service Tax:")
                    Text(String.format(Locale.US, "RM %.2f", taxCents / 100.0))
                }
                Divider(modifier = Modifier.padding(vertical = 8.dp))
                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
                    Text("Grand Total", style = MaterialTheme.typography.headlineSmall)
                    Text(formattedTotal, style = MaterialTheme.typography.headlineSmall)
                }
            }
        }

        Text("Select Payment Method", style = MaterialTheme.typography.titleMedium)
        Spacer(modifier = Modifier.height(12.dp))

        // --- ENABLED PAYMENT METHODS GRID ---
        LazyVerticalGrid(
            columns = GridCells.Fixed(2),
            modifier = Modifier.weight(1f),
            horizontalArrangement = Arrangement.spacedBy(12.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            items(availableMethods) { method ->
                val isSelected = selectedMethod == method
                Card(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(80.dp)
                        .clickable { selectedMethod = method }
                        .border(
                            width = if (isSelected) 3.dp else 1.dp,
                            color = if (isSelected) MaterialTheme.colorScheme.primary else Color.LightGray,
                            shape = CardDefaults.shape
                        )
                ) {
                    Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                        Text(
                            text = method.displayName,
                            style = MaterialTheme.typography.bodyMedium,
                            color = if (isSelected) MaterialTheme.colorScheme.primary else Color.Unspecified
                        )
                    }
                }
            }
        }

        // --- SUBMIT PAYMENT ACTION ---
        Button(
            enabled = selectedMethod != null && !isProcessing,
            onClick = {
                selectedMethod?.let { method ->
                    isProcessing = true
                    // Launch payment call using PaymentGatewayClient...
                }
            },
            modifier = Modifier.fillMaxWidth().height(56.dp)
        ) {
            if (isProcessing) {
                CircularProgressIndicator(color = Color.White)
            } else {
                Text("Pay $formattedTotal", style = MaterialTheme.typography.titleMedium)
            }
        }
    }
}
